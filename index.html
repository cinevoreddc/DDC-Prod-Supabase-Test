-- DDC PROD EXPENSES MANAGEMENT — ETAPE 11B -- Avances complementaires sur une demande DEJA validee et decaissee. -- A executer apres l'etape 11A, dans Supabase > SQL Editor. -- SAUVEGARDE : exporter vos donnees financieres AVANT la migration. -- Le script n'effectue aucun mouvement bancaire ou de Mobile Money. -- Aucun montant de demande originale ni justification deja approuvee n'est efface. -- TEST : n'utiliser que des dossiers fictifs. begin; create table if not exists public.complements_financement ( id uuid primary key default gen_random_uuid(), demande_id text not null references public.demandes(id) on delete restrict, montant_ariary bigint not null check (montant_ariary > 0 and montant_ariary <= 1000000000000), motif text not null check (char_length(trim(motif)) between 10 and 500), demande_par text not null references public.utilisateurs(id), demande_le timestamptz not null default now(), statut text not null default 'en_attente' check (statut in ('en_attente','validee','refusee','decaissee')), decide_par text references public.utilisateurs(id), decide_le timestamptz, commentaire_decision text check (commentaire_decision is null or char_length(commentaire_decision) <= 500), decaisse_par text references public.utilisateurs(id), decaisse_le timestamptz, mode_paiement text check (mode_paiement is null or mode_paiement in ('Espèces','Virement','Mobile Money','Autre')), reference_paiement text check (reference_paiement is null or char_length(reference_paiement) <= 120), check (decide_par is null or decide_par <> demande_par), check (statut = 'en_attente' or decide_par is not null), check (statut <> 'decaissee' or (decaisse_par is not null and decaisse_le is not null and mode_paiement is not null)) ); create index if not exists idx_ddc_complements_demande on public.complements_financement(demande_id, demande_le desc); -- Une seule avance complementaire en attente/validee (non encore payee) par demande. -- Refusee ou decaissee : on peut deposer une nouvelle demande complementaire. create unique index if not exists uq_ddc_complement_en_cours on public.complements_financement(demande_id) where statut in ('en_attente','validee'); alter table public.complements_financement enable row level security; revoke all on public.complements_financement from public, anon, authenticated; grant select on public.complements_financement to authenticated; -- Prepare une extension ulterieure du reporting via la fonction de service, sans exposer les donnees anonymement. grant select on public.complements_financement to service_role; drop policy if exists ddc_complements_read on public.complements_financement; create policy ddc_complements_read on public.complements_financement for select to authenticated using ( (select private.ddc_is_finance()) or exists ( select 1 from public.demandes d where d.id = complements_financement.demande_id and d.demandeur_id = (select private.ddc_my_id()) ) ); -- A. Demande : l'auteur proprietaire sollicite une rallonge (sans paiement automatique). create or replace function private.ddc_request_supplement( p_request text, p_amount bigint, p_reason text ) returns uuid language plpgsql volatile security definer set search_path = '' as $ddc$ declare v_actor text; v_owner text; v_paid bigint; v_ref text := pg_catalog.btrim(coalesce(p_request, '')); v_reason text := pg_catalog.btrim(coalesce(p_reason, '')); v_result uuid; begin if (select auth.uid()) is null then raise exception 'Connexion requise.' using errcode = '28000'; end if; select u.id into v_actor from public.utilisateurs u where u.auth_id=(select auth.uid()) and u.actif=true; if v_actor is null then raise exception 'Compte DDC non autorise.' using errcode = '42501'; end if; if p_amount is null or p_amount<=0 or p_amount>1000000000000 then raise exception 'Montant complementaire invalide.'; end if; if pg_catalog.char_length(v_reason)<10 or pg_catalog.char_length(v_reason)>500 then raise exception 'Le motif doit compter entre 10 et 500 caracteres.'; end if; -- Toutes les operations financieres verrouillent d'abord la demande. select d.demandeur_id, d.montant_decaisse_ariary into v_owner,v_paid from public.demandes d where d.id=v_ref and d.validation='validee' and d.decaissement='decaisse' and d.controle<>'conforme' for update; if not found or v_owner<>v_actor then raise exception 'Seul le demandeur peut demander une avance sur son dossier ouvert et decaisse.' using errcode = '42501'; end if; if v_paid is null or v_paid<=0 or p_amount>1000000000000-v_paid then raise exception 'Le cumul du decaissement et du complement depasserait la limite.'; end if; if exists(select 1 from public.complements_financement c where c.demande_id=v_ref and c.statut in ('en_attente','validee')) then raise exception 'Un complement est deja en attente de validation ou de decaissement.'; end if; insert into public.complements_financement(demande_id,montant_ariary,motif,demande_par) values(v_ref,p_amount,v_reason,v_actor) returning id into v_result; insert into public.historique(demande_id,acteur_id,action,commentaire) values(v_ref,v_actor,'complement_demande', 'Avance complementaire demandee : '||p_amount::text||' Ar · '||v_reason); return v_result; end; $ddc$; create or replace function public.ddc_demander_complement( p_demande text, p_montant bigint, p_motif text ) returns uuid language sql volatile security invoker set search_path = '' as $ddc$ select private.ddc_request_supplement($1,$2,$3); $ddc$; -- B. Validation ou refus : Roger ou Mamihasina, pas de validation personnelle. create or replace function private.ddc_decide_supplement( p_supplement uuid, p_decision text, p_comment text ) returns uuid language plpgsql volatile security definer set search_path = '' as $ddc$ declare v_actor text; v_request text; v_owner text; v_status text; v_amount bigint; v_decision text := pg_catalog.btrim(coalesce(p_decision,'')); v_comment text := pg_catalog.btrim(coalesce(p_comment,'')); begin if (select auth.uid()) is null then raise exception 'Connexion requise.' using errcode='28000'; end if; select u.id into v_actor from public.utilisateurs u where u.auth_id=(select auth.uid()) and u.actif=true and u.role in ('finance','daf'); if v_actor is null then raise exception 'Validation reservee a Roger ou Mamihasina.' using errcode='42501'; end if; if p_supplement is null or v_decision not in ('validee','refusee') then raise exception 'Decision ou identifiant invalide.'; end if; if pg_catalog.char_length(v_comment)>500 or (v_decision='refusee' and pg_catalog.char_length(v_comment)<5) then raise exception 'Commentaire invalide (motif de refus : 5 caracteres minimum, 500 maximum).'; end if; -- Verrou d'abord sur demande, puis complement pour eviter les courses. select c.demande_id into v_request from public.complements_financement c where c.id=p_supplement; if v_request is null then raise exception 'Avance complementaire introuvable.'; end if; select d.demandeur_id into v_owner from public.demandes d where d.id=v_request and d.validation='validee' and d.decaissement='decaisse' and d.controle<>'conforme' for update; if not found or v_owner=v_actor then raise exception 'Dossier ferme, invalide ou autovalidation interdite.' using errcode='42501'; end if; select c.statut,c.montant_ariary into v_status,v_amount from public.complements_financement c where c.id=p_supplement and c.demande_id=v_request for update; if not found or v_status<>'en_attente' then raise exception 'Cette demande complementaire a deja recu une decision.'; end if; update public.complements_financement c set statut=v_decision,decide_par=v_actor,decide_le=pg_catalog.now(), commentaire_decision=nullif(v_comment,'') where c.id=p_supplement; insert into public.historique(demande_id,acteur_id,action,commentaire) values(v_request,v_actor, case when v_decision='validee' then 'complement_valide' else 'complement_refuse' end, 'Complement '||v_decision||' : '||v_amount::text||' Ar'|| case when v_comment<>'' then ' · '||v_comment else '' end); return p_supplement; end; $ddc$; create or replace function public.ddc_decider_complement( p_complement uuid, p_decision text, p_commentaire text ) returns uuid language sql volatile security invoker set search_path = '' as $ddc$ select private.ddc_decide_supplement($1,$2,$3); $ddc$; -- C. Decaissement : tout montant complementaire entre en solde seulement ici. create or replace function private.ddc_disburse_supplement( p_supplement uuid, p_mode text, p_reference text ) returns uuid language plpgsql volatile security definer set search_path = '' as $ddc$ declare v_actor text; v_request text; v_owner text; v_status text; v_amount bigint; v_paid bigint; v_control text; v_proof text; v_mode text := pg_catalog.btrim(coalesce(p_mode,'')); v_ref text := pg_catalog.btrim(coalesce(p_reference,'')); begin if (select auth.uid()) is null then raise exception 'Connexion requise.' using errcode='28000'; end if; select u.id into v_actor from public.utilisateurs u where u.auth_id=(select auth.uid()) and u.actif=true and u.role in ('finance','daf'); if v_actor is null then raise exception 'Decaissement reserve a Roger ou Mamihasina.' using errcode='42501'; end if; if p_supplement is null or v_mode not in ('Espèces','Virement','Mobile Money','Autre') then raise exception 'Reference ou mode de paiement invalide.'; end if; if pg_catalog.char_length(v_ref)>120 then raise exception 'Reference de paiement trop longue.'; end if; select c.demande_id into v_request from public.complements_financement c where c.id=p_supplement; if v_request is null then raise exception 'Avance complementaire introuvable.'; end if; select d.demandeur_id,d.montant_decaisse_ariary,d.controle,d.justificatif into v_owner,v_paid,v_control,v_proof from public.demandes d where d.id=v_request and d.validation='validee' and d.decaissement='decaisse' and d.controle<>'conforme' for update; if not found or v_owner=v_actor then raise exception 'Dossier ferme, invalide ou autodecaissement interdit.' using errcode='42501'; end if; select c.statut,c.montant_ariary into v_status,v_amount from public.complements_financement c where c.id=p_supplement and c.demande_id=v_request for update; if not found or v_status<>'validee' then raise exception 'L avance complementaire doit etre validee et non encore decaissee.'; end if; if v_paid is null or v_paid<=0 or v_amount>1000000000000-v_paid then raise exception 'Montant total decaisse excessif.'; end if; update public.demandes d set montant_decaisse_ariary=v_paid+v_amount, controle=case when v_control='a_corriger' and v_proof='envoye' then 'a_verifier' else d.controle end, controle_par=case when v_control='a_corriger' and v_proof='envoye' then null else d.controle_par end, controle_le=case when v_control='a_corriger' and v_proof='envoye' then null else d.controle_le end, commentaire_controle=case when v_control='a_corriger' and v_proof='envoye' then null else d.commentaire_controle end where d.id=v_request; update public.complements_financement c set statut='decaissee',decaisse_par=v_actor,decaisse_le=pg_catalog.now(), mode_paiement=v_mode,reference_paiement=nullif(v_ref,'') where c.id=p_supplement; insert into public.historique(demande_id,acteur_id,action,commentaire) values(v_request,v_actor,'complement_decaisse', 'Avance complementaire payee : '||v_amount::text||' Ar · '||v_mode|| case when v_ref<>'' then ' · Ref. '||v_ref else '' end|| ' · Total decaisse : '||(v_paid+v_amount)::text||' Ar'); return p_supplement; end; $ddc$; create or replace function public.ddc_decaisser_complement( p_complement uuid,p_mode_paiement text,p_reference text ) returns uuid language sql volatile security invoker set search_path = '' as $ddc$ select private.ddc_disburse_supplement($1,$2,$3); $ddc$; -- Blocage serveur de la cloture Conforme si une avance attend une decision ou un paiement. -- Le controle existant de l'etape 11A reste intact et continue a calculer : -- montants justifies + fonds restitues = total decaisse (y compris complement verse). create or replace function private.ddc_block_closure_with_pending_supplements() returns trigger language plpgsql volatile security definer set search_path = '' as $ddc$ begin if new.controle='conforme' and old.controle is distinct from new.controle and exists( select 1 from public.complements_financement c where c.demande_id=new.id and c.statut in ('en_attente','validee') ) then raise exception 'Cloture impossible : une avance complementaire attend validation ou decaissement.'; end if; return new; end; $ddc$; drop trigger if exists ddc_guard_pending_complements on public.demandes; create trigger ddc_guard_pending_complements before update of controle on public.demandes for each row execute function private.ddc_block_closure_with_pending_supplements(); revoke all on function private.ddc_decide_supplement(uuid,text,text) from public,anon,authenticated; revoke all on function private.ddc_disburse_supplement(uuid,text,text) from public,anon,authenticated; revoke all on function private.ddc_block_closure_with_pending_supplements() from public,anon,authenticated; revoke all on function public.ddc_demander_complement(text,bigint,text) from public,anon,authenticated; revoke all on function public.ddc_decider_complement(uuid,text,text) from public,anon,authenticated; revoke all on function public.ddc_decaisser_complement(uuid,text,text) from public,anon,authenticated; grant execute on function public.ddc_demander_complement(text,bigint,text) to authenticated; grant execute on function public.ddc_decider_complement(uuid,text,text) to authenticated; grant execute on function public.ddc_decaisser_complement(uuid,text,text) to authenticated;
